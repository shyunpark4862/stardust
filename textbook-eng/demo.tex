The symbols used throughout this volume are collected here. Each is defined again where it first appears.
\symbolgroup{Sets and numbers}
\begin{symbols}
\item[$\mathbb Z$, $\mathbb R$] the integers and the real numbers
\item[$\mathbb F$] the set of binary64 floating-point numbers
\item[{$[a,b)$}] the interval $\{x\in\mathbb R: a\le x<b\}$
\end{symbols}
\symbolgroup{Floating-point arithmetic}
\begin{symbols}
\item[$\fl(x)$] the binary64 number nearest to the real number $x$, ties to even
\item[$\oplus$, $\ominus$] floating-point addition and subtraction, $a\oplus b=\fl(a+b)$
\item[$u$] the unit roundoff, $2^{-53}$
\item[$\operatorname{ulp}(x)$] the spacing of binary64 numbers at $x$
\end{symbols}
